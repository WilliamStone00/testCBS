public void buildAndSaveBankDebitRequestLogOnRedisStream(String internalId, String msisdn, RequestAutoDebitDto requestAutoDebitDto){
        var loggerObject2 = LoggerData.builder()
                .exception("")
                .internalId(internalId)
                .kpiName("Carry over")
                .objectData(dataUtils.extractJsonStringFromObject(requestAutoDebitDto))
                .kpiValue("Bank debit request")
                .clientStatus("Registered")
                .operation("LOAN")
                .serviceName("TFJ")
                .orderOfExecution("1")
                .serviceExecutionState("send the request to ecw to bankDebit the client")
                .status("Continue")
                .msisdn(msisdn)
                .build();
        writeInRedisLogStream(loggerObject2);
    }

public void writeInRedisLogStream(LoggerData loggerData){
        LocalDateTime date = LocalDateTime.now();

        Map<String, String> map = new HashMap<String, String>();

        map.put("internalId",loggerData.getInternalId());
        map.put("msisdn",loggerData.getMsisdn());
        map.put("dateTime",date+"");
        map.put("operation",loggerData.getOperation());
        map.put("serviceName",loggerData.getServiceName());
        map.put("serviceExecutionState",loggerData.getServiceExecutionState());
        map.put("clientStatus",loggerData.getClientStatus());
        map.put("status",loggerData.getStatus());
        map.put("orderOfExecution",loggerData.getOrderOfExecution());
        map.put("objectData",loggerData.getObjectData());
        map.put("exception",loggerData.getException());
        map.put("kpiName",loggerData.getKpiName());
        map.put("kpiValue",loggerData.getKpiValue());


        ObjectRecord<String, Map<String, String>> record = StreamRecords.newRecord()
                .ofObject(map)
                .withStreamKey(StreamNames.EVENT_LOG_STREAM_NAME);

        RecordId recordId = redisTemplateCache.opsForStream().add(record);
        assert recordId != null;
    }
