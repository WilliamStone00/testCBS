            function escapeHtml(s) {
                return (s || "").replace(/[&<>"']/g, function (c) {
                    return ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]);

            function pick(o, camel, pascal, fallback) {
                if (!o) return fallback;
                if (o[camel] !== undefined && o[camel] !== null) return o[camel];
                if (o[pascal] !== undefined && o[pascal] !== null) return o[pascal];
                return fallback;
            }

            function inRange(n, min, max) {
                n = Number(n);
                min = Number(min);
                max = Number(max);
                if (isNaN(n)) return false;
                if (!isNaN(min) && min > 0 && n < min) return false;
                if (!isNaN(max) && max > 0 && n > max) return false;
                return true;
            }

            function validateInterestToPolicy() {
                var policy = window.productPolicy || {};
                var minI = Number(policy.minInterest || 0);
                var maxI = Number(policy.maxInterest || 0);

                var $i = $("#txtInterest");
                var raw = ($i.val() || "").toString().trim();
                var v = parseFloat(raw);

                // empty => let your submit required checks handle it
                if (raw === "") {
                    $i.removeClass("is-invalid");
                    return true;
                }

                if (isNaN(v) || v <= 0) {
                    $i.addClass("is-invalid");
                    ensureInvalidFeedback($i, "Enter a valid interest rate.");
                    return false;
                }

                var ok = (v >= minI) && (v <= maxI);
                if (!ok) {
                    $i.addClass("is-invalid");
                    ensureInvalidFeedback($i, "Interest must be within product range (" + minI + "% - " + maxI + "%).");

                    // keep focus on interest so user fixes it
                    setTimeout(function () { $i.focus(); }, 0);
                    return false;
                }

                $i.removeClass("is-invalid");
                return true;
            }

                        bindRepaymentPeriods({ Periods: [] });
                        return dfd.resolve(null).promise();
                    });

            function autoSelectPolicyRequiredAccounts() {
                // Guard
                if (!window.productPolicy) return;
                if (!Array.isArray(recoveryAllAccounts)) return;

                // Loan amount
                var loanAmount = Number($("#txtAmount").val());
                if (isNaN(loanAmount) || loanAmount < 0) loanAmount = 0;

                // Policy values
                var p = window.productPolicy;

                // Required amounts:
                // Savings required = max(absolute min, % of amount)
                var savingsAbs = Number(p.minSavingsAbs || 0);
                var savingsPct = Number(p.minSavingsPercent || 0);
                var savingsByPct = (loanAmount > 0 && savingsPct > 0) ? (loanAmount * (savingsPct / 100)) : 0;
                var savingsRequired = Math.max(savingsAbs, savingsByPct);

                // Shares required = policy min shares required (already max(product, policy))
                var sharesRequired = Number(p.minSharesRequired || 0);

                // Helpers
                function normStr(v) { return (v == null) ? "" : String(v).trim(); }

                function findFirstAccount(predicate) {
                    for (var i = 0; i < recoveryAllAccounts.length; i++) {
                        if (predicate(recoveryAllAccounts[i])) return recoveryAllAccounts[i];
                    }
                    return null;
                }

                function isSavings(a) {
                    var t = normStr(pick(a, "accountType", "AccountType", "")).toLowerCase();
                    t = t.replace(/\s+/g, " ").trim();
                    return t === "savings";
                }

                function isOrdinaryShares(a) {
                    var t = normStr(pick(a, "accountType", "AccountType", "")).toLowerCase();
                    t = t.replace(/\s+/g, " ").trim();
                    return t === "ordinary shares"; // strict
                }

                function loadMemberAccountsForRecovery() {
                    if (!accountsUrl || !wizardState.memberId) return;

                    $("#recoveryLoadHint").text("Loading accounts...");

                    $.get(accountsUrl, { memberId: wizardState.memberId })
                        .done(function (res) {
                            var items = (res && (res.data || res.Data)) || [];
                            recoveryAllAccounts = items;

                            autoSelectPolicyRequiredAccounts();
                            renderRecoveryPicker();
                            renderRecoveryEditor();
                            //syncRecoveryHiddenJson();

                            $("#recoveryLoadHint").text(items.length ? "" : "No accounts found.");
                        })
                        .fail(function () {
                            $("#recoveryLoadHint").text("Failed to load accounts.");
                            console.warn("Accounts failed to load.");
                        });
                }


                // Find member accounts
                var savingsAcc = findFirstAccount(isSavings);
                var sharesAcc = findFirstAccount(isOrdinaryShares);

                // Decide if policy requires them:
                // ✅ You said both are required for loan, so enforce if loanAmount > 0
                var requireSavings = (loanAmount > 0) && (savingsRequired > 0);
                var requireShares = (loanAmount > 0) && (sharesRequired > 0);

                // Apply to selection map
                function upsertPolicyAccount(acc, requiredAmount, reasonLabel) {
                    if (!acc) return;
                    var id = normStr(pick(acc, "id", "Id", ""));
                    if (!id) return;

                    recoverySelected[id] = recoverySelected[id] || {};

                    recoverySelected[id].amount = Math.ceil(Number(requiredAmount || 0)); // or keep as is
                    recoverySelected[id].decision = "RequiredPolicy"; // internal key
                    recoverySelected[id].locked = true;
                    recoverySelected[id].policyReason = reasonLabel || "Required by policy";
                    recoverySelected[id].policyType = "REQUIRED";
                }

                // If required -> add/lock
                if (requireSavings) {
                    upsertPolicyAccount(
                        savingsAcc,
                        savingsRequired,
                        "Savings required by policy (" +
                        (savingsPct > 0 ? (savingsPct + "%") : "") +
                        (savingsAbs > 0 ? (savingsPct > 0 ? " and/or " : "") + "min " + savingsAbs.toLocaleString() + " FCFA" : "") +
                        ")."
                    );
                }

                if (requireShares) {
                    upsertPolicyAccount(
                        sharesAcc,
                        sharesRequired,
                        "Ordinary Shares required by policy (min " + sharesRequired.toLocaleString() + " FCFA). Not blockable."
                    );
                }

                // If not required anymore (amount cleared), unlock/remove ONLY if it was policy-managed
                function dropIfPolicyManaged(acc) {
                    if (!acc) return;
                    var id = normStr(pick(acc, "id", "Id", ""));
                    if (!id) return;
                    if (recoverySelected[id] && recoverySelected[id].locked) {
                        delete recoverySelected[id];
                    }
                }

                if (!requireSavings) dropIfPolicyManaged(savingsAcc);
                if (!requireShares) dropIfPolicyManaged(sharesAcc);

                // Re-render
                renderRecoveryPicker();

            function feeTypeBadge(calcType) {
                // calcType might be int OR string; normalize
                var t = String(calcType || "").toUpperCase();
                if (t === "1" || t === "FIXED") return `<span class="badge bg-success-subtle text-success border border-success-subtle">Fixed</span>`;
                if (t === "2" || t === "RATE") return `<span class="badge bg-info-subtle text-info border border-info-subtle">Rate</span>`;
                if (t === "3" || t === "RANGE") return `<span class="badge bg-purple-subtle text-purple border border-purple-subtle">Range</span>`;
                return `<span class="badge bg-light text-dark border">Unknown</span>`;

            function computeChargeValue(rule, amount) {
                amount = toNum(amount);
                if (!amount || amount <= 0) return 0;

                var t = normalizeCalcType(rule); // ✅ FIXED/RATE/RANGE (supports int and string)

                if (t === "FIXED") {
                    return toNum(rule.FixedAmount ?? rule.fixedAmount ?? 0);
                }

                if (t === "RATE") {
                    var rate = toNum(rule.RatePercent ?? rule.ratePercent ?? 0);
                    var raw = (amount * rate / 100.0);

                    var min = toNum(rule.MinFee ?? rule.minFee ?? 0);
                    var max = toNum(rule.MaxFee ?? rule.maxFee ?? 0);

                    if (min > 0) raw = Math.max(min, raw);
                    if (max > 0) raw = Math.min(max, raw);

                    return Math.round(raw);
                }

                if (t === "RANGE") {
                    var ranges = rule.Ranges || rule.ranges || [];
                    for (var i = 0; i < ranges.length; i++) {
                        var r = ranges[i];
                        var from = toNum(r.FromAmount ?? r.fromAmount ?? 0);
                        var to = toNum(r.ToAmount ?? r.toAmount ?? 0);
                        var fee = toNum(r.FeeAmount ?? r.feeAmount ?? 0);

                        if (amount >= from && amount <= to) return fee;
                    }
                    return 0;
                }

                return 0;
            }

            function calcTypeDetailText(rule) {
                var t = normalizeCalcType(rule);

                function toNum(v) { v = Number(v); return isNaN(v) ? 0 : v; }

                if (t === "RATE") {
                    var rate = toNum(rule.RatePercent ?? rule.ratePercent);
                    var minF = toNum(rule.MinFee ?? rule.minFee);
                    var maxF = toNum(rule.MaxFee ?? rule.maxFee);

                    var parts = [rate + "%"];
                    if (minF > 0) parts.push("Min " + minF.toLocaleString());
                    if (maxF > 0) parts.push("Max " + maxF.toLocaleString());
                    return parts.join(" • ");
                }

                if (t === "RANGE") {
                    var ranges = (rule.Ranges ?? rule.ranges ?? []);
                    if (!ranges.length) return "No tiers";
                    // show short summary
                    var first = ranges[0];
                    var last = ranges[ranges.length - 1];
                    var from = Number(first.FromAmount ?? first.fromAmount ?? 0).toLocaleString();
                    var to = Number(last.ToAmount ?? last.toAmount ?? 0).toLocaleString();
                    return "Tiers (" + from + " → " + to + ")";
                }

                if (t === "FIXED") {
                    var fixed = Number(rule.FixedAmount ?? rule.fixedAmount ?? 0);
                    return fixed.toLocaleString() + " FCFA";
                }

                return "";
            }
            function computeSystemCharge(charge, amount) {
                amount = Number(amount || 0);
                var type = String(charge.CalcType || charge.calcType || "").toUpperCase();

                function toNum(v) { v = Number(v); return isNaN(v) ? 0 : v; }

                if (type === "FIXED") {
                    return toNum(charge.FixedAmount || charge.fixedAmount);
                }

                if (type === "RANGE") {
                    var ranges = (charge.Ranges || charge.ranges || []);
                    for (var i = 0; i < ranges.length; i++) {
                        var r = ranges[i];
                        var from = toNum(r.FromAmount || r.fromAmount);
                        var to = toNum(r.ToAmount || r.toAmount);
                        var fee = toNum(r.FeeAmount || r.feeAmount);

                        // inclusive range
                        if (amount >= from && amount <= to) return fee;
                    }
                    return 0;
                }

                if (type === "RATE") {
                    var rate = toNum(charge.RatePercent || charge.ratePercent); // percent
                    var minF = toNum(charge.MinFee || charge.minFee);
                    var maxF = toNum(charge.MaxFee || charge.maxFee);

                    var fee = (amount * rate) / 100.0;
                    if (minF > 0) fee = Math.max(fee, minF);
                    if (maxF > 0) fee = Math.min(fee, maxF);

                    return Math.round(fee); // optional rounding
                }

                return 0;
            }
            function normalizeCalcType(rule) {
                var t = (rule.CalcType ?? rule.calcType ?? "").toString().trim();

                // int enum support: 1 Fixed, 2 Rate, 3 Range
                if (t === "1") return "FIXED";
                if (t === "2") return "RATE";
                if (t === "3") return "RANGE";

                // string support
                t = t.toUpperCase();
                if (t.includes("FIX")) return "FIXED";
                if (t.includes("RATE") || t.includes("%")) return "RATE";
                if (t.includes("RANGE") || t.includes("TIER")) return "RANGE";

                return "UNKNOWN";
            }

            function inRangeNumber(v, min, max) {
                if (v === null || v === undefined || isNaN(v)) return false;
                if (min !== null && min !== undefined && !isNaN(min) && v < min) return false;
                if (max !== null && max !== undefined && !isNaN(max) && v > max) return false;
                return true;
            }

                if (v !== undefined && v !== null) return v;

            function applyRequiredRecoveryAccounts() {
                window.recoveryState = window.recoveryState || {};
                window.recoveryState.selected = window.recoveryState.selected || {};

                var all = window.recoveryState.accounts || [];
                var selected = window.recoveryState.selected;

                all.forEach(function (a) {
                    if (!isPolicyRequiredAccount(a)) return;

                    var id = String(pick(a, "id", "Id", ""));
                    var bal = toNum(pick(a, "balance", "Balance", 0));

                    // Decide policy behavior:
                    // - OrdinaryShares required => locked + not blockable + fixed min amount from API if provided
                    // - Savings required => locked + blocked (or required) depending on your rules

                    var minAmt = toNum(pick(a, "minRequiredAmount", "MinRequiredAmount", 0)); // ✅ best: backend supplies
                    var requiredAmt = minAmt > 0 ? minAmt : Math.min(bal, bal); // fallback: keep existing

                    // Preserve existing if already selected
                    selected[id] = selected[id] || {};

                    selected[id].locked = true;
                    selected[id].decision = isOrdinaryShares(a) ? "Required" : "Blocked"; // shares required not blockable -> keep "Required"
                    selected[id].amount = requiredAmt;
                    selected[id].reason = pick(a, "policyMessage", "PolicyMessage", "Required by policy.");
                });
            }

            function statusBadge(status) {

                if (s === "DORMANT") return `<span class="badge bg-warning-subtle text-warning border">DORMANT</span>`;

            function requirePolicyReady() {
                return !!(window.productPolicy && (window.productPolicy.minSharesRequired !== undefined));
            }
            /**
             * ✅ Apply current policy => auto-select & lock Savings/Shares rules
             * - Shares: required, not blockable => Decision=Required, Amount=0, locked
             * - Savings: required => Decision=Blocked, Amount=max(minSavingsAbs, loanAmount * minSavingsPercent/100), locked
             */
            function applyRecoveryPolicyFromProductPolicy() {
                if (!window.recoveryState.enabled) return;
                if (!requirePolicyReady()) return;

                var p = window.productPolicy;
                var loanAmount = getLoanAmount();

                var minSavingsAbs = toNum(p.minSavingsAbs);
                var minSavingsPercent = toNum(p.minSavingsPercent);
                var minSharesRequired = toNum(p.minSharesRequired);

                var needSavings = (minSavingsAbs > 0) || (minSavingsPercent > 0);
                var needShares = (minSharesRequired > 0);

                var items = window.recoveryState.accounts || [];
                var selected = window.recoveryState.selected || {};

                // ✅ helper: create/update a selected entry
                function upsert(id, patch) {
                    selected[id] = selected[id] || { amount: 0, decision: "KeepOpen", locked: false, reason: "" };
                    Object.assign(selected[id], patch);
                }

                // ✅ helper: remove policy entry if policy no longer requires it
                function dropPolicy(id, policyKey) {
                    if (!selected[id]) return;
                    if (selected[id].locked && selected[id].policyKey === policyKey) delete selected[id];
                }

                items.forEach(function (a) {
                    var id = String(pick(a, "id", "Id", ""));
                    if (!id) return;

                    // ✅ Ordinary Shares (Required, not blockable)
                    if (needShares && isOrdinaryShares(a)) {
                        upsert(id, {
                            decision: "Required",
                            amount: minSharesRequired,
                            locked: true,
                            reason: `Shares required by policy (min ${minSharesRequired.toLocaleString()} FCFA). Not blockable.`,
                            policyKey: "SHARES_REQUIRED"
                        });
                        return;
                    }

                    // ✅ Savings (Blocked, auto amount)
                    if (needSavings && isSavings(a)) {
                        var pctAmt = Math.round((loanAmount * minSavingsPercent) / 100);
                        var amt = Math.max(minSavingsAbs, pctAmt);

                        upsert(id, {
                            decision: "Blocked",
                            amount: amt,
                            locked: true,
                            reason: `Savings required by policy (${minSavingsPercent}% and/or min ${minSavingsAbs.toLocaleString()} FCFA).`,
                            policyKey: "SAVINGS_REQUIRED"
                        });
                        return;
                    }

                    // ✅ IMPORTANT: do NOTHING for other accounts (do NOT create selected entry)
                });

                // If policy no longer requires them, remove only the locked policy-managed entries
                if (!needSavings) {
                    items.filter(isSavings).forEach(a => dropPolicy(String(pick(a, "id", "Id", "")), "SAVINGS_REQUIRED"));
                }
                if (!needShares) {
                    items.filter(isOrdinaryShares).forEach(a => dropPolicy(String(pick(a, "id", "Id", "")), "SHARES_REQUIRED"));
                }

                window.recoveryState.selected = selected;

                renderRecoveryPicker();
                renderRecoveryEditor();
                //syncRecoveryHiddenJson?.();
            }

            }

                        $amt.prop("disabled", false);

                    buildRecoverySubmissionJson();
                });

                    var $i = $(this);

                    var acc = recoveryAccountsCache.find(x => String(x.Id || x.id) === id);

                    if (v > bal) {
                        $i.addClass("is-invalid");
                        v = bal;
                        $i.val(v);
                    } else {
                        $i.removeClass("is-invalid");
                    }

                    recoverySelections[id] = recoverySelections[id] || { decision: "Blocked", amount: 0 };
                    recoverySelections[id].amount = v;

                    buildRecoverySubmissionJson();
                });

            function calcRecoveryTotals() {
                var selected = window.recoveryState.selected || {};

                var totalUserSelected = 0;          // sum of all selected amounts (including required)
                var totalBlocked = 0;               // sum where decision implies blocked/lien/required
                var totalPolicyRequired = 0;        // sum of locked required amounts only

                Object.keys(selected).forEach(function (id) {
                    var x = selected[id];
                    var amt = toNum(x.amount);

                    totalUserSelected += amt;

                    if (x.locked) totalPolicyRequired += amt;

                    var d = upper(x.decision);
                    var isBlockish = (d === "BLOCKED" || d === "LIEN" || d === "REQUIRED");
                    if (isBlockish) totalBlocked += amt;
                });

                var coob = getCoobligorTotal();
                var shotee = getShoteeTotal();

                var downPct = getDownpaymentPercent();
                var requested = getRequestedLoanAmount();
                var downRequired = Math.round((requested * downPct) / 100);

                var provided = totalUserSelected + coob + shotee;
                var covered = provided >= downRequired;
                var remaining = Math.max(0, downRequired - provided);

                return {
                    requested, downPct, downRequired,
                    totalUserSelected, totalPolicyRequired,
                    coob, shotee,
                    provided, covered, remaining,
                    totalBlocked
                };
            }

            function requiredScoreByAmount(amount) {
                amount = Number(amount || 0);
                if (!amount || amount <= 0) return 0;

                var cfg = scoreConfig;
                var bands = cfg && (cfg.bands || cfg.Bands);

                if (!bands || !bands.length) {
                    // fallback (only if server failed)
                    if (amount <= 200000) return 40;
                    if (amount <= 1000000) return 60;
                    return 75;
                }

                for (var i = 0; i < bands.length; i++) {
                    var b = bands[i];
                    var max = Number(b.maxAmount || b.MaxAmount || 0);
                    var req = Number(b.requiredScore || b.RequiredScore || 0);
                    if (max > 0 && amount <= max) return req;
                }

                var last = bands[bands.length - 1];
                return Number(last.requiredScore || last.RequiredScore || 0);
            }
            function suggestedMaxAmountByScore(score) {
                score = Number(score || 0);
                var cfg = scoreConfig;
                var bands = cfg && (cfg.bands || cfg.Bands);
                if (!bands || !bands.length) return 0;

                // Sort by maxAmount ascending (safe)
                var sorted = bands.slice().sort(function (a, b) {
                    return Number(a.maxAmount || a.MaxAmount || 0) - Number(b.maxAmount || b.MaxAmount || 0);
                });

                var best = 0;
                for (var i = 0; i < sorted.length; i++) {
                    var b = sorted[i];
                    var max = Number(b.maxAmount || b.MaxAmount || 0);
                    var req = Number(b.requiredScore || b.RequiredScore || 0);
                    if (max > 0 && score >= req) best = max;
                }
                return best;
            }

            function computeBlockedByKind() {
                var blocks = (window.tscRecovery && window.tscRecovery.getBlocks)

                for (var i = 0; i < blocks.length; i++) {
                    var b = blocks[i];
                    var amt = parseFloat(b.BlockAmount) || 0;
                    var label = labelMap[b.AccountId] || "";

            }
            function inferAccountKindFromText(text) {
                var t = String(text || "").toLowerCase();
                // You can adjust keywords to match your labels
                if (t.includes("ordinary shares") || t.includes("shares") || t.includes("shr-")) return "SHARES";
                if (t.includes("savings") || t.includes("sav-")) return "SAVINGS";
                return "OTHER";
            }



            function evaluateScore() {
                try {
                    var amount = Number($("#txtAmount").val() || 0);

                    // no amount => hide result
                    if (!amount || amount <= 0) {
                        $("#scoreResultBox").hide();
                        return;
                    }

                    // points from selected assessments
                    var score = 0;
                    $("#ddlScoreAssessments option:selected").each(function () {
                        var pts = Number($(this).attr("data-points") || 0); // safer than .data() with dynamic options
                        score += isNaN(pts) ? 0 : pts;
                    });

                    var required = requiredScoreByAmount(amount);

                    var overBonus = scoreConfig ? Number(scoreConfig.overScoreBonus || scoreConfig.OverScoreBonus || 15) : 15;
                    var medBuf = scoreConfig ? Number(scoreConfig.mediumRiskBuffer || scoreConfig.MediumRiskBuffer || 10) : 10;
                    var suggested = suggestedMaxAmountByScore(score);
                    var policy = window.productPolicy || {};
                    var policyMax = Number(policy.maxAmount || 0);

                    if (suggested > 0 && policyMax > 0) {
                        suggested = Math.min(suggested, policyMax);
                    }

                    var requested = amount;

                    var suggestionText = "";
                    if (suggested > 0) {
                        if (requested > suggested) {
                            suggestionText =
                                "Suggested amount based on score: " + suggested.toLocaleString() + " FCFA. " +
                                "Tip: reduce requested amount to meet scoring comfort.";
                        } else {
                            suggestionText =
                                "Suggested amount based on score: up to " + suggested.toLocaleString() + " FCFA.";
                        }
                    }
                    if (suggested > 0 && policyMax > 0 && suggested === policyMax) {
                        suggestionText += " (Capped by product max amount)";
                    }

                    // Put this text somewhere (add one small line in UI)
                    $("#scoreSuggestedAmount").text(suggestionText);

                    // risk bands
                    var status, badgeClass, advice;

                    if (score >= required + overBonus) {
                        status = "NO RISK - Loan is covered";
                        badgeClass = "bg-success";
                        advice = "Recommendation: Proceed. Member is over-scored for this amount. Keep standard mitigation only.";
                    } else if (score >= required) {
                        status = "LOW RISK";
                        badgeClass = "bg-primary";
                        advice = "Recommendation: Proceed with normal controls. Consider light mitigation if policy requires.";
                    } else if (score >= required - medBuf) {
                        status = "MEDIUM RISK";
                        badgeClass = "bg-warning text-dark";
                        advice = "Recommendation: Reduce amount OR add stronger mitigation (blocking/guarantor/collateral) before approval.";
                    } else {
                        status = "HIGH RISK";
                        badgeClass = "bg-danger";
                        advice = "Recommendation: Do not proceed unless exceptional mitigation is added OR the amount is reduced significantly.";
                    }

                    $("#scoreResultBox").show();
                    $("#scoreBadge").attr("class", "badge rounded-pill " + badgeClass).text(status);
                    $("#scoreValue").text(score);
                    $("#scoreRequired").text(required);
                    $("#scoreAdvice").text(advice);

                    // delta
                    var delta = score - required;
                    $("#scoreDelta")
                        .text((delta >= 0 ? "+" : "") + delta + " pts")
                        .toggleClass("text-success", delta >= 0)
                        .toggleClass("text-danger", delta < 0);

                } catch (e) {
                    console.error("evaluateScore failed:", e);
                }
            }

            function getRequestedAmountLive() {
                return toNum($("#txtAmount").val() || 0);
            }

            function getDownpaymentRateFromPolicy() {
                var p = window.productPolicy || {};
                // prefer explicit dp rate
                var r = toNum(p.downPaymentRatePercent || p.DownPaymentRatePercent || 0);
                return r;
            }

            // Provided = sum of all selected coobligor/shotee blocked amounts + any other selected recovery accounts
            // NOTE: you said: "sum of selected accounts excluding ordinaryshares must be the % of amount requested"
            // We'll compute using recoveryState.selected if present, ignoring ordinary shares.
            function computeDownpaymentProvidedFromRecovery() {
                var provided = 0;

                // Recovery accounts selection (your existing state)
                var rec = (window.recoveryState && window.recoveryState.selected) ? window.recoveryState.selected : {};
                var accounts = (window.recoveryState && window.recoveryState.accounts) ? window.recoveryState.accounts : [];

                Object.keys(rec).forEach(function (id) {
                    var sel = rec[id] || {};
                    var amt = toNum(sel.amount);
                    if (amt <= 0) return;

                    // exclude Ordinary Shares
                    var acc = accounts.find(a => String(pick(a, "id", "Id", "")) === String(id));
                    var t = (acc ? String(pick(acc, "accountType", "AccountType", "")) : "").toLowerCase().replace(/\s+/g, " ").trim();
                    if (t === "ordinary shares" || t === "ordinaryshares") return;

                    provided += amt;
                });

                // Also add from guarantor entries for COOBLIGOR/SHOTEE if they block amounts (stored in guarantor state)
                var gsum = 0;
                (riskState.guarantors || []).forEach(function (g) {
                    var type = String(g.guarantorType || "").toUpperCase();
                    if (type === "COOBLIGOR" || type === "SHOTEE" || type === "INTER_COOPERATION") {
                        gsum += toNum(g.amount || 0);
                    }
                });

                return provided + gsum;
            }

            function renderDownpaymentBox() {
                var requested = getRequestedAmountLive();
                var rate = getDownpaymentRateFromPolicy();
                var required = Math.round((requested * rate) / 100);

                var provided = computeDownpaymentProvidedFromRecovery();
                var remaining = Math.max(0, required - provided);
                var covered = (provided >= required);

                $("#dpRequestedAmt").text(requested.toLocaleString());
                $("#dpRate").text(rate);
                $("#dpRequiredAmt").text(required.toLocaleString());
                $("#dpProvidedAmt").text(provided.toLocaleString());
                $("#dpRemainingAmt").text(remaining.toLocaleString());

                $("#dpCoverageBadge")
                    .removeClass()
                    .addClass("badge border " + (covered ? "bg-success-subtle text-success" : "bg-danger-subtle text-danger"))
                    .text(covered ? "COVERED" : "NOT COVERED");

                riskState.downpayment = {
                    enabled: true, rate: rate, required: required,
                    provided: provided, remaining: remaining, covered: covered,
                    note: $("#txtDownPaymentNote").val() || ""
                };

                $("#hidDownPaymentJson").val(JSON.stringify(riskState.downpayment));
            }

            function rebuildGuarantorsJson() {
                // normalize from DOM
                var arr = [];
                $("#guarantorList [data-gid]").each(function () {
                    var $c = $(this);
                    var gid = $c.data("gid");

                    var type = String($c.find(".ddlGuarantorType").val() || "").toUpperCase();
                    var isMemberBacked = (type === "COOBLIGOR" || type === "SHOTEE" || type === "INTER_COOPERATION");

                    var item = {
                        id: gid,
                        guarantorType: type,
                        memberReference: $c.find(".txtMemberRef").val() || "",
                        memerberName: $c.find(".txtNameCompany").val() || "",
                        companyOrGroupName: $c.find(".txtNameCompany").val() || "",
                        affiliateOrInstitutionName: $c.find(".txtNameCompany").val() || "",

                        idCardNumber: $c.find(".txtIdCardNumber").val() || "",
                        issueDate: $c.find(".txtIssueDate").val() || "",
                        expireDate: $c.find(".txtExpireDate").val() || "",

                        relationship: $c.find(".txtRelationship").val() || "",
                        address: $c.find(".txtAddress").val() || "",
                        phoneNumber: $c.find(".txtPhone").val() || "",
                        email: $c.find(".txtEmail").val() || "",

                        guaranteeAmount: toNum($c.find(".txtGuaranteeAmount").val() || 0),

                        // for coobligor/shotee/intercoop
                        accountNumber: isMemberBacked ? ($c.find(".ddlAccount").val() || "") : "",
                        amount: isMemberBacked ? toNum($c.find(".txtBlockAmount").val() || 0) : 0,
                        blockAccount: isMemberBacked ? String($c.find(".ddlBlockFlag").val() || "true") === "true" : false,

                        // release order (assigned later)
                        releaseOrder: null
                    };

                    arr.push(item);
                });

                // release order assignment if needed (only for blocking member-backed entries)
                var blockables = arr.filter(x =>
                    (x.guarantorType === "COOBLIGOR" || x.guarantorType === "SHOTEE" || x.guarantorType === "INTER_COOPERATION") &&
                    x.blockAccount === true && toNum(x.amount) > 0
                );

                // require release order only if more than 1
                $("#releaseOrderWrap").toggle(blockables.length > 1);

                if (blockables.length > 1) {
                    // build UI
                    var html = "";
                    blockables.forEach(function (x) {
                        html += `
  <div class="col-12 col-md-6" data-rid="${escHtml(x.id)}">
    <div class="border rounded p-2">
      <div class="fw-semibold small">${escHtml(x.memerberName || x.memberReference || x.id)}</div>
      <div class="text-muted small">Account: ${escHtml(x.accountNumber || "-")} • Block: ${toNum(x.amount).toLocaleString()} FCFA</div>
      <label class="form-label small text-muted mb-1 mt-2">Release Order</label>
      <select class="form-select form-select-sm ddlReleaseOrder">
        <option value="">-- Select --</option>
        ${blockables.map((_, i) => `<option value="${i + 1}">${i + 1}</option>`).join("")}
      </select>
    </div>
  </div>
`;
                    });
                    $("#releaseOrderList").html(`<div class="row g-2">${html}</div>`);

                    // read values back
                    $("#releaseOrderList .ddlReleaseOrder").off("change.rel").on("change.rel", function () {
                        var $p = $(this).closest("[data-rid]");
                        var rid = String($p.data("rid"));
                        var v = $(this).val() ? Number($(this).val()) : null;

                        arr.forEach(function (a) {
                            if (String(a.id) === rid) a.releaseOrder = v;
                        });

                        riskState.guarantors = arr;
                        $("#hidGuarantorsJson").val(JSON.stringify(arr));
                        if ($("#chkDownPayment").is(":checked")) renderDownpaymentBox();
                    });
                } else {
                    $("#releaseOrderList").empty();
                }

                riskState.guarantors = arr;
                $("#hidGuarantorsJson").val(JSON.stringify(arr));

                if ($("#chkDownPayment").is(":checked")) renderDownpaymentBox();
            }

                        $card.find(".txtMemberRef").val("");
                        $card.find(".memberSearchHint").text("");
                    }

                    rebuildGuarantorsJson();
                });

                // changes -> rebuild
                $card.find("input, select, textarea").off("input.gAny change.gAny").on("input.gAny change.gAny", function () {
                    rebuildGuarantorsJson();
                });

                // member search (coobligor/shotee/intercoop)
                $card.find(".btnSearchMember").off("click.searchM").on("click.searchM", function () {
                    var memberRef = String($card.find(".txtMemberRef").val() || "").trim();
                    if (!memberRef) {
                        $card.find(".memberSearchHint").text("Enter a member reference.");
                        return;
                    }
                    if (!memberLookupUrl) {
                        $card.find(".memberSearchHint").text("Member lookup URL not configured.");
                        return;
                    }

                    $card.find(".memberSearchHint").html(`<span class="text-muted">Searching...</span>`);

                    // Expected response: { success:true, data:{ memberId, name, idCardNumber, phone, email, address, issueDate, expireDate } }
                    $.get(memberLookupUrl, { memberReference: memberRef })
                        .done(function (res) {
                            var ok = (res && (res.success === true || res.Success === true));
                            var data = (res && (res.data || res.Data)) || null;

                            if (!ok || !data) {
                                $card.find(".memberSearchHint").html(`<span class="text-danger">Member not found.</span>`);
                                return;
                            }

                            // Fill KYC
                            $card.find(".txtNameCompany").val(data.name || data.Name || "");
                            $card.find(".txtIdCardNumber").val(data.idCardNumber || data.IdCardNumber || "");
                            $card.find(".txtPhone").val(data.phoneNumber || data.PhoneNumber || "");
                            $card.find(".txtEmail").val(data.email || data.Email || "");
                            $card.find(".txtAddress").val(data.address || data.Address || "");
                            $card.find(".txtIssueDate").val(data.issueDate || data.IssueDate || "");
                            $card.find(".txtExpireDate").val(data.expireDate || data.ExpireDate || "");

                            $card.data("memberId", data.memberId || data.MemberId || "");
                            $card.find(".memberSearchHint").html(`<span class="text-success">Member loaded.</span>`);

                            // Load accounts
                            var mid = $card.data("memberId");
                            if (!mid || !memberAccountsUrl) return;

                            $card.find(".ddlAccount").html(`<option value="">Loading accounts...</option>`);
                            $.get(memberAccountsUrl, { memberId: mid })
                                .done(function (ares) {
                                    var items = (ares && (ares.data || ares.Data)) || [];
                                    var opts = `<option value="">-- Select Account --</option>`;

                                    items.forEach(function (a) {
                                        var accNo = pick(a, "accountNumber", "AccountNumber", "");
                                        var type = pick(a, "accountType", "AccountType", "");
                                        var bal = toNum(pick(a, "balance", "Balance", 0));
                                        var cur = pick(a, "currency", "Currency", "FCFA") || "FCFA";
                                        opts += `<option value="${escHtml(accNo)}">${escHtml(type)} - ${escHtml(accNo)} (Bal: ${bal.toLocaleString()} ${escHtml(cur)})</option>`;
                                    });

                                    $card.find(".ddlAccount").html(opts);
                                    rebuildGuarantorsJson();
                                })
                                .fail(function () {
                                    $card.find(".ddlAccount").html(`<option value="">Failed to load accounts</option>`);
                                });

                            rebuildGuarantorsJson();
                        })
                        .fail(function () {
                            $card.find(".memberSearchHint").html(`<span class="text-danger">Lookup failed.</span>`);
                        });
                });

            function syncMitigation(name, isRequired) {
                // Naming convention example:
                // checkbox: #chkGuarantor
                // wrapper:  #wrapGuarantor
                // add btn:  #btnAddGuarantor
                var id = name.replace(/\s/g, "");

                var $chk = $("#chk" + id);
                var $wrap = $("#wrap" + id);

                if (!$chk.length || !$wrap.length) return;

                if (isRequired) {
                    $chk.prop("checked", true).prop("disabled", true); // user cannot uncheck required
                    $wrap.show(); // show panel so user can click "Add"
                } else {
                    $chk.prop("disabled", false);
                    // respect current user choice
                    if ($chk.is(":checked")) $wrap.show(); else $wrap.hide();
                }
            }

            // user toggles (optional ones)
            $(document).on("change", ".riskMitigationToggle", function () {
                var name = $(this).data("name"); // e.g. data-name="Guarantor"
                syncMitigation(name, false);     // not required, so just follow checked/unchecked
            });

            function loadMemberAccountsIntoGuarantorCard($card, memberId) {
                var routes = getRoutes();
                if (!routes.memberAccountsUrl) {
                    console.warn("memberAccountsUrl missing in #loanUiRoutes");
                    return;
                }
                if (!memberId) return;

                $.getJSON(routes.memberAccountsUrl, { memberId: memberId })
                    .done(function (res) {
                        if (!res || !res.success) return;

                        var $ddl = $card.find(".ddlAccount");
                        if (!$ddl.length) return;

                        $ddl.empty().append('<option value="">-- Select account --</option>');

                        (res.data || []).forEach(function (a) {
                            var txt = (a.AccountType || "") + " - " + (a.AccountNumber || "") + " | " + (a.AccountName || "") + " | Bal: " + (a.Balance || 0) + " " + (a.Currency || "");
                            $ddl.append('<option value="' + a.Id + '" data-type="' + (a.AccountType || "") + '" data-balance="' + (a.Balance || 0) + '">' + txt + '</option>');
                        });

                        // Show account block UI now that accounts exist
                        $card.find(".wrapAccountSelect,.wrapBlockAmount,.wrapBlockFlag").show();
                    })
                    .fail(function (xhr) {
                        console.error("GetMemberAccountsByMemberId failed:", xhr);
                    });
            }
